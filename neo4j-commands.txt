# MASUK KE NEO4J 
RETURN n10s.version() AS version;

# untuk inisialisasi konfigurasi rdf
CALL n10s.graphconfig.init({
  handleVocabUris: "SHORTEN",
  handleMultival: "ARRAY",
  keepLangTag: true,
  keepCustomDataTypes: true
});

# buat constraint
CREATE CONSTRAINT n10s_unique_uri IF NOT EXISTS
FOR (r:Resource) REQUIRE r.uri IS UNIQUE;

# import ttl 
CALL n10s.rdf.import.fetch(
  "file:///var/lib/neo4j/import/graph000004_000001.ttl",
  "Turtle"
);

# buat vector index baru
CREATE VECTOR INDEX vector IF NOT EXISTS
FOR (n:Resource) ON (n.embedding)
OPTIONS {
  indexConfig: {
      `vector.dimensions`: 384,
      `vector.similarity_function`: 'cosine'
  }
};

# buat keyword index
CREATE FULLTEXT INDEX keyword IF NOT EXISTS
FOR (n:Resource) ON EACH [n.name, n.description];

# cek list index
SHOW INDEXES YIELD name, type, state;
